# Python dependencies for home-monitoring.
#
# IMPORTANT: The Jetson base image (dustynv/jetson-inference, JetPack 4.x)
# ships Python 3.6. All versions must therefore be Python-3.6-compatible
# (no current FastAPI/uvicorn/pytest releases).
# The base image already provides: torch 1.10 (CUDA), opencv 4.5 (CUDA+dnn), GStreamer.

# Web stack (Python 3.6-compatible)
fastapi==0.68.2
uvicorn==0.16.0

# dataclasses backport: Python 3.6 does not have the module in the stdlib (since 3.7).
dataclasses==0.6

# ffmpeg binary for event recording (static, aarch64).
# The base image contains no ffmpeg; imageio-ffmpeg provides a matching
# binary (found automatically in app/recorder.py).
imageio-ffmpeg==0.4.9

# Object detection runs via onnxruntime + ONNX (app/detector.py).
# cv2.dnn (OpenCV 4.5) cannot parse YOLOv8 ONNX; onnxruntime 1.10.0
# is the last version with a Python-3.6 + aarch64 wheel.
# Model: yolov8n.onnx in the ./models directory (see README).
onnxruntime==1.10.0

# Tests
pytest==7.0.1
